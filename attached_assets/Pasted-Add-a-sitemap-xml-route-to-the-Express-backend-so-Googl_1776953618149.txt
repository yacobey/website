Add a sitemap.xml route to the Express backend so 
Google can discover all pages.

In server/routes.ts add this endpoint:

app.get('/sitemap.xml', (req, res) => {
  const baseUrl = 'https://selamcpa.com';
  const pages = [
    { url: '/', priority: '1.0', changefreq: 'weekly' },
    { url: '/tax', priority: '0.9', changefreq: 'monthly' },
    { url: '/bookkeeping', priority: '0.9', changefreq: 'monthly' },
    { url: '/advisory', priority: '0.9', changefreq: 'monthly' },
    { url: '/audit', priority: '0.9', changefreq: 'monthly' },
    { url: '/ai-consulting', priority: '0.9', changefreq: 'monthly' },
    { url: '/calculators', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/scorp-savings', priority: '0.9', changefreq: 'monthly' },
    { url: '/calculators/estimated-tax', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/self-employment-tax', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/home-office', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/mileage', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/section-179', priority: '0.8', changefreq: 'monthly' },
    { url: '/calculators/retirement', priority: '0.8', changefreq: 'monthly' },
    { url: '/tools', priority: '0.7', changefreq: 'monthly' },
    { url: '/industries/healthcare', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/legal', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/real-estate', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/technology', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/retail', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/construction', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/nonprofit', priority: '0.8', changefreq: 'monthly' },
    { url: '/industries/hospitality', priority: '0.8', changefreq: 'monthly' },
    { url: '/blog', priority: '0.7', changefreq: 'weekly' },
    { url: '/contact', priority: '0.7', changefreq: 'monthly' },
  ];

  const today = new Date().toISOString().split('T')[0];
  
  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${pages.map(page => `  <url>
    <loc>${baseUrl}${page.url}</loc>
    <lastmod>${today}</lastmod>
    <changefreq>${page.changefreq}</changefreq>
    <priority>${page.priority}</priority>
  </url>`).join('\n')}
</urlset>`;

  res.header('Content-Type', 'application/xml');
  res.send(xml);
});

Also add a robots.txt route:

app.get('/robots.txt', (req, res) => {
  res.type('text/plain');
  res.send(`User-agent: *
Allow: /
Sitemap: https://selamcpa.com/sitemap.xml`);
});
