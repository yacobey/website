const API = import.meta.env.VITE_API_URL as string;

    const price_id = "price_TAX_PREP_123"; // TODO: replace

  return (

          <button onClick={startTaxPrep} className="btn btn-primary">Start Tax Prep</button>
        </div>
        <p className="phone muted">📞 <a href={`tel:${phone.replace(/[^\d+]/g, "")}`}>{phone}</a> — Virtual Receptionist (24/7)</p>
      </section>

      <section aria-labelledby="tax-strategy">
