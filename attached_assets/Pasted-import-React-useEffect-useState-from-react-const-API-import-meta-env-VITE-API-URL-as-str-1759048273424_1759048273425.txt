import React, { useEffect, useState } from "react";

const API = import.meta.env.VITE_API_URL as string;
const AGENT_URL = import.meta.env.VITE_AGENT_PUBLIC_URL as string;

export default function Agent() {
  const [pro, setPro] = useState(false);
  const [loading, setLoading] = useState(true);

  async function fetchStatus() {
    setLoading(true);
    const r = await fetch(`${API}/agent/status`, { credentials: "include" });
    const data = await r.json();
    setPro(!!data.pro);
    setLoading(false);
  }

  useEffect(() => { fetchStatus(); }, []);

  async function subscribe() {
    const r = await fetch(`${API}/subscribe-agent`, { method: "POST", credentials: "include" });
    const data = await r.json();
    if (data.url) window.location.href = data.url;
  }
  async function openBilling() { window.location.href = `${API}/agent/billing`; }
  async function logoutPro() {
    await fetch(`${API}/agent/logout`, { method: "POST", credentials: "include" });
    fetchStatus();
  }

  return (
    <main className="wrap">
      <h1>SelamTax CPA Agent</h1>

      {loading ? (
        <div className="alert">Checking subscription…</div>
      ) : !pro ? (
        <div className="alert warn">
          <strong>Free Mode:</strong> Brief summaries only.{" "}
          <button onClick={subscribe} className="btn btn-primary" style={{ marginLeft: 8 }}>Subscribe to Agent Pro</button>
        </div>
      ) : (
        <div className="alert ok">
          <strong>Agent Pro:</strong> Full, actionable answers unlocked.
          <button className="btn" onClick={openBilling} style={{ marginLeft: 8 }}>Manage Subscription</button>
          <button className="btn" onClick={logoutPro} style={{ marginLeft: 8 }}>Sign out</button>
        </div>
      )}

      <p className="muted">If the agent doesn’t load, <a href={AGENT_URL} target="_blank" rel="noopener">open it in a new tab</a>.</p>
      <iframe className="agent-iframe" src={AGENT_URL} title="SelamTax CPA Agent" />
      <section className="mt-16 muted" style={{ fontSize: 14 }}>
        <p><strong>Privacy note:</strong> Don’t upload documents here. Use our secure upload link after checkout.</p>
      </section>
    </main>
  );
}
